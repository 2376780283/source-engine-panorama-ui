// Sound emitter manifest for the panorama test mod.
//
// CSoundEmitterSystemBase::BaseInit() (soundemittersystem/soundemittersystembase.cpp) Error()s out
// when this file is missing, and KeyValues refuses to load a manifest whose root block has no
// sub-keys at all.  The loader accepts "precache_file", "preload_file" and "faceposer_file".
//
// game_sounds_ui_panorama.txt is CS:GO's UI sound script: the panorama content dispatches
// PlaySoundEffect("UIPanorama.<name>") for every button rollover/press/popup, and the sound system
// resolves those names through the sound emitter system (CUISoundSystem::PlaySound +
// panorama_s1wrapper/wrap_sound.cpp).  Without this entry the script is never parsed, every lookup
// fails and the whole UI stays silent.
"game_sounds_manifest"
{
	"precache_file"		"scripts/game_sounds_ui_panorama.txt"
}
